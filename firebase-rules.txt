====================================================================
REGRAS DO FIRESTORE - Cloud Firestore > Rules
====================================================================

Cole estas regras no Console do Firebase:
Firebase Console > Firestore Database > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Admin (pode tudo)
    function isAdmin() {
      return request.auth != null && request.auth.token.email == 'brayan@storsistemas.com.br';
    }

    // Perfis: todos leem, dono ou admin escreve
    match /profiles/{userId} {
      allow read: if true;
      allow create: if request.auth != null && request.auth.uid == userId;
      allow update: if request.auth != null && (request.auth.uid == userId || isAdmin());
      allow delete: if request.auth != null && (request.auth.uid == userId || isAdmin());
    }

    // Atividades: só o dono ou admin
    match /activities/{docId} {
      allow read: if request.auth != null && (resource.data.user_id == request.auth.uid || isAdmin());
      allow create: if request.auth != null && request.data.user_id == request.auth.uid;
      allow update: if request.auth != null && (get(/databases/$(database)/documents/activities/$(docId)).data.user_id == request.auth.uid || isAdmin());
      allow delete: if request.auth != null && (get(/databases/$(database)/documents/activities/$(docId)).data.user_id == request.auth.uid || isAdmin());
    }
  }
}
